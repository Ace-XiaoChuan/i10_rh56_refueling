cmake_minimum_required(VERSION 3.16)
project(observed_rviz_capture)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(capture capture.cpp)
target_compile_features(capture PUBLIC cxx_std_17)
ament_target_dependencies(capture rclcpp rviz_common rviz_rendering)
target_link_libraries(capture Qt5::Widgets rviz_common::rviz_common rviz_rendering::rviz_rendering)
