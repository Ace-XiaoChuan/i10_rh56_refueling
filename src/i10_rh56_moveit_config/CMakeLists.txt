cmake_minimum_required(VERSION 3.16)
project(i10_rh56_moveit_config)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(geometric_shapes REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(moveit_task_constructor_core REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(audit_model src/audit_model.cpp)
target_compile_features(audit_model PUBLIC cxx_std_17)
target_compile_options(audit_model PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(audit_model rclcpp moveit_core moveit_ros_planning moveit_msgs geometric_shapes)
target_link_libraries(audit_model yaml-cpp)
add_executable(plan_approach src/plan_approach.cpp)
target_compile_features(plan_approach PUBLIC cxx_std_17)
ament_target_dependencies(plan_approach rclcpp moveit_core moveit_ros_planning moveit_msgs geometric_shapes moveit_task_constructor_core moveit_ros_planning_interface)
target_link_libraries(plan_approach yaml-cpp nlohmann_json::nlohmann_json)
install(TARGETS plan_approach DESTINATION lib/${PROJECT_NAME})
install(TARGETS audit_model DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config meshes launch DESTINATION share/${PROJECT_NAME})
ament_package()
